# ── 3. AIRFLOW (GitHub Codespace — Linux) ──────────────────────

# Vérifier la version
airflow version

# Initialiser la base Airflow (Airflow 3.x)
airflow db migrate

# Lancer Airflow en mode standalone (tout en un)
AIRFLOW__WEBSERVER__BASE_URL="https://<codespace-name>-8080.app.github.dev" airflow standalone

# Trouver le nom du codespace
echo $CODESPACE_NAME

# Trouver le mot de passe admin généré
cat ~/airflow/simple_auth_manager_passwords.json.generated

# URL d'accès à l'interface web
# https://<CODESPACE_NAME>-8080.app.github.dev


# ── 4. CONFIGURATION DU DAG ────────────────────────────────────

# Créer le dossier dags
mkdir -p ~/airflow/dags

# Copier le DAG
cp /workspaces/projetFinalTovo/projetFinal/elia/dags/weather_etl_dag.py ~/airflow/dags/

# Corriger les imports dépréciés (Airflow 3.x)
sed -i 's/from airflow.operators.python import PythonOperator/from airflow.providers.standard.operators.python import PythonOperator/' ~/airflow/dags/weather_etl_dag.py

# Corriger le chemin des modules Python
sed -i 's|BASE_DIR = os.path.dirname(os.path.dirname(os.path.abspath(__file__)))|BASE_DIR = "/workspaces/projetFinalTovo/projetFinal/elia"|' ~/airflow/dags/weather_etl_dag.py

# Corriger schedule_interval → schedule (Airflow 3.x)
sed -i 's/schedule_interval=/schedule=/g' ~/airflow/dags/weather_etl_dag.py

# Tester le DAG sans erreur
python ~/airflow/dags/weather_etl_dag.py

# Lister les DAGs détectés
airflow dags list
airflow dags list | grep weather

# Activer les DAGs
airflow dags unpause weather_current_dag
airflow dags unpause weather_daily_dag

# Déclencher un DAG manuellement
airflow dags trigger weather_daily_dag


# ── 5. POSTGRESQL (Codespace) ──────────────────────────────────

# Installer PostgreSQL
sudo apt-get update && sudo apt-get install -y postgresql postgresql-contrib

# Démarrer le service
sudo service postgresql start
